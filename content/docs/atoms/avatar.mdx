## Usage

```tsx
import { Avatar, AvatarFallback, AvatarImage } from '@/components/atoms/avatar'
```

```tsx
<Avatar>
  <AvatarImage src='https://github.com/shadcn.png' />
  <AvatarFallback>CN</AvatarFallback>
</Avatar>
```

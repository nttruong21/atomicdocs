## Usage

```tsx
import {
  DatePicker,
  type DatePickerProps,
} from '@/components/atoms/date-picker'
```

```tsx
const [value, setValue] = useState<DatePickerProps['value']>(null)

<DatePicker className='w-xs' onValueChange={setValue} placeholder='Select date' value={value} />
```

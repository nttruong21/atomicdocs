## Usage

```tsx
import {
  FileUpload,
  FileUploadContent,
  FileUploadInput,
  FileUploadItem,
  type FileUploadProps,
} from '@/components/molecules/file-upload'
```

```tsx
const [value, setValue] = useState<FileUploadProps['value']>([])

<FileUpload
  value={value}
  dropzoneOptions={{
    maxFiles: 5
  }}
  onValueChange={setValue}
>
  <FileUploadInput />
  <FileUploadContent>
    {value.map((item, index) => (
      <FileUploadItem key={`${index}-${item instanceof File ? item.name : item.id}`} value={item} index={index} />
    ))}
  </FileUploadContent>
</FileUpload>
```

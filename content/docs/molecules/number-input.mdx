## Usage

```tsx
import {
  NumberInput,
  type NumberInputProps,
} from '@/components/molecules/number-input'
```

```tsx
const [value, setValue] = useState<NumberInputProps['value']>('')

<NumberInput
  value={value}
  placeholder='Enter number'
  onFieldChange={setValue}
  onValueChange={({ value }) => setValue(value)}
/>
```

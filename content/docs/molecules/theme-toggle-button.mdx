## Usage

```tsx
import React from 'react'
import {
  observeTheme,
  ThemeToggleButton,
} from '@/components/molecules/theme-toggle-button'
```

```tsx
React.useLayoutEffect(() => {
  const observer = observeTheme('theme')
  return () => observer.disconnect()
}, [])

<ThemeToggleButton themeLocalStorageKey='theme' />
```

## Usage

### Client Side Pagination

```tsx
import { type ColumnDef, getPaginationRowModel } from '@tanstack/react-table'
import { DataTable } from '@/components/organisms/data-table/data-table'
import { useDataTable } from '@/components/organisms/data-table/lib'
```

```tsx
const table = useDataTable({
  columns,
  data,
  getPaginationRowModel: getPaginationRowModel()
})

<DataTable table={table} />
```

### Server Side Pagination

```tsx
import { type ColumnDef, type PaginationState } from '@tanstack/react-table'
import { DataTable } from '@/components/organisms/data-table/data-table'
import { useDataTable } from '@/components/organisms/data-table/lib'
```

```tsx
const rowCount = useState(0)
const [pagination, setPagination] = useState<PaginationState>({
  pageIndex: 0,
  pageSize: 10
})

const table = useDataTable({
  columns,
  data,
  manualPagination: true,
  rowCount,
  state: { pagination },
  onPaginationChange: setPagination
})

<DataTable table={table} />
```

## Composition

Use the following composition to build a `DataTable`:

```text
table.AppTable
├── table.ColumnVisibilitySelection
└── table.Container
    ├── table.Table
    │   ├── table.Header
    │   ├── table.Body
    │   └── table.Footer
    ├── table.AdditionalInfo
    ├── table.RowSelection
    └── table.Pagination
```
